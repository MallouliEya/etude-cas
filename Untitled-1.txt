    <script>
        document.getElementById('travelOfferForm').addEventListener('submit', function(event) {
            event.preventDefault(); // Empêche la soumission du formulaire

            // Réinitialiser les messages d'erreur
            document.querySelectorAll('.error-message').forEach(el => el.textContent = '');

            const title = document.getElementById("title").value;
            const destination = document.getElementById("destination").value;
            const departureDate = document.getElementById("departure_date").value;
            const returnDate = document.getElementById("return_date").value;
            const price = document.getElementById("price").value;

            let isValid = true;

            if (title.length < 3) {
                document.getElementById('titleError').textContent = "The title must contain at least 3 characters.";
                isValid = false;
            }

            if (destination.length < 3 || !/^[a-zA-Z\s]+$/.test(destination)) {
                document.getElementById('destinationError').textContent = "The destination must contain only letters and at least 3 characters.";
                isValid = false;
            }

            const departureDateObj = new Date(departureDate);
            const returnDateObj = new Date(returnDate);

            if (isNaN(departureDateObj) || isNaN(returnDateObj)) {
                document.getElementById('departureDateError').textContent = "Please select valid dates.";
                document.getElementById('returnDateError').textContent = "Please select valid dates.";
                isValid = false;
            } else if (returnDateObj <= departureDateObj) {
                document.getElementById('returnDateError').textContent = "Return date must be later than departure date.";
                isValid = false;
            }

            if (isNaN(price) || price <= 0) {
                document.getElementById('priceError').textContent = "The price must be a positive number.";
                isValid = false;
            }

            if (isValid) {
                // Si tout est valide, vous pouvez soumettre le formulaire ici
                // par exemple: this.submit(); ou envoyer les données via AJAX
                alert("Formulaire valide !"); // Ceci est un exemple, à remplacer par votre logique
            }
        });
    </script>
